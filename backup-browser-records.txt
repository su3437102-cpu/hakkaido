// 在正在使用的北海道网页控制台运行；只导出北海道应用的浏览器记录。
(()=>{const records={};for(let i=0;i<localStorage.length;i++){const k=localStorage.key(i);if(/^hk-(route-map-|must-pack-|v2-plan$)/.test(k))records[k]=localStorage.getItem(k);}const url=URL.createObjectURL(new Blob([JSON.stringify({format:'hokkaido-browser-backup',version:1,created:new Date().toISOString(),origin:location.origin,records},null,2)],{type:'application/json'}));const a=document.createElement('a');a.href=url;a.download='hokkaido-records.json';a.click();setTimeout(()=>URL.revokeObjectURL(url),1000);console.log('已导出',Object.keys(records).length,'项北海道记录');})();
